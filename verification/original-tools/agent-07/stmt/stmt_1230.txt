def Frontier.GateC : Prop :=
∃ P,
  P.Exact ∧
    ∃ T,
      (∀ (G : Frontier.Graph) (s : Fin G.n), P.RunsWithin G s (T G.n G.m)) ∧
        ∃ μ,
          (∀ᶠ (n : ℕ) in Filter.atTop, ↑n * √(Real.log (↑n + 2)) ≤ ↑(μ n) ∧ ↑(μ n) ≤ ↑n * Real.log (↑n + 2)) ∧
            Filter.Tendsto (fun n => T n (μ n) / (↑n * Real.log (↑n + 2))) Filter.atTop (nhds 0)
def Frontier.GateCTarget.CHDTarget : (ℕ → ℕ) → Prop :=
fun F =>
  ∃ P, P.Exact ∧ ∃ C, ∀ (G : Frontier.Graph) (s : Fin G.n), P.RunsWithin G s (C * Frontier.GateCTarget.Tdisp F G.n G.m)
def Frontier.GateCCalc.Tchd : ℕ → ℕ → ℝ :=
fun n m => ↑n + ↑m + ↑m * Real.log (↑m / (↑n + 1) + 2) + ↑m ^ (1 / 3) * (↑n * Real.log (↑n + 2)) ^ (2 / 3)
def Frontier.GateCCalc.F : ℕ → ℕ :=
fun n => (Nat.log 2 n ^ 3).sqrt.sqrt
Frontier.GateCTarget.chdTarget_F_imp_gateC : Frontier.GateCTarget.CHDTarget Frontier.GateCCalc.F → Frontier.GateC
Frontier.CHD.Dispatch.chdTarget_of_body : ∀ (e : ℕ),
  3 ≤ e →
    ∀ (ps : List Frontier.RAM.Stmt) (chd : Frontier.RAM.Stmt) (C₁ : ℝ),
      0 ≤ C₁ →
        (Frontier.CHD.Dispatch.BodySpec e ps chd (fun G => G.m ≤ G.n * Frontier.GateCCalc.F G.n ∧ 16 ≤ Nat.log 2 G.n)
            fun n m => C₁ * Frontier.GateCCalc.Tchd n m) →
          Frontier.GateCTarget.CHDTarget Frontier.GateCCalc.F
